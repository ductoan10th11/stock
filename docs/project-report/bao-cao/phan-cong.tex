\section*{3. Nguyên tắc làm việc chung}

\begin{itemize}
  \item Mã nguồn quản lý bằng Git; mỗi thành viên commit bằng tài khoản của chính mình, commit nhỏ và thông điệp rõ nghĩa.
  \item Mọi đường dẫn trong mã tính tương đối so với thư mục gốc của kho, cố định \texttt{random\_\allowbreak state = 42}.
  \item Số liệu trong báo cáo lấy từ mã chạy được của kho, không ghi tay.
  \item Phần sử dụng AI tạo sinh được khai báo trong Lời cam đoan.
\end{itemize}
